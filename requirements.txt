# Python 3.12
# Install:  pip install -r requirements.txt
#
# Versions are pinned exactly. The results reported in the accompanying paper
# were produced with this environment.
#
# scipy is pinned explicitly even though it arrives as a scikit-learn
# dependency. It supplies the LAPACK bindings used by LinearRegression, and the
# design matrix here is close to rank deficient (condition number ~4.9e7), so
# the linear regression baseline shifts slightly between scipy builds. The other
# four models are unaffected.
#
# Build the virtualenv on a standalone Python, not on a conda interpreter.
# A venv layered over conda inherits conda's OpenMP/BLAS libraries, which clash
# with the ones bundled in the TensorFlow and XGBoost wheels and segfault on
# import (macOS especially).

pandas==3.0.5
scipy==1.18.0
scikit-learn==1.9.0
xgboost==3.3.0
tensorflow==2.21.0
keras==3.15.0
